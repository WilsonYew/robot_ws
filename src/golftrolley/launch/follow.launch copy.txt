from launch import LaunchDescription
from launch.actions import IncludeLaunchDescription, TimerAction
from launch.launch_description_sources import PythonLaunchDescriptionSource
from launch_ros.actions import Node, ComposableNodeContainer
from launch_ros.substitutions import FindPackageShare
from launch.substitutions import PathJoinSubstitution
from launch_ros.descriptions import ComposableNode
from ament_index_python.packages import get_package_share_directory
import os

def generate_launch_description():

    # disparity_launch = Node(
    #         package='stereo_image_proc', executable='disparity_node',
    #         parameters=[{'approximate_sync': True}],
    #         remappings=[
    #             ('left/image_rect',  '/left/image_rect'),
    #             ('right/image_rect', '/right/image_rect'),
    #             ('left/camera_info','/left/camera_info'),
    #             ('right/camera_info','/right/camera_info'),
    #             ('disparity',       '/stereo/disparity'),
    #         ],
    #         respawn=True
    #     )

    disparity = Node(
        package='stereo_image_proc',
        executable='disparity_node',
        name='disparity',
        remappings=[
            ('left/image_rect',  '/left/image_rect'),
            ('right/image_rect', '/right/image_rect'),
            ('left/camera_info','/left/camera_info'),
            ('right/camera_info','/right/camera_info'),
            ('disparity',        '/disparity'),
        ],
        parameters=[{
            'approximate_sync': True,
        }],
        output='screen'
    )

    point_cloud = Node(
        package='stereo_image_proc',
        executable='point_cloud_node',
        name='points2',
        remappings=[
            ('left/image_rect',  '/left/image_rect_rgb'),
            ('left/camera_info','/left/camera_info'),
            ('disparity',        '/disparity'),
            ('points2',          '/left/points2'),
        ],
        parameters=[{
            'approximate_sync': True,
        }],
        output='screen'
    )



    pkg = 'golftrolley'
    collision_params = os.path.join(get_package_share_directory(pkg), 'config', 'collision_monitor.yaml')
    smoother_params = os.path.join(get_package_share_directory(pkg), 'config', 'nav2_velocity_smoother.yaml')
    follow_params = os.path.join(get_package_share_directory(pkg), 'config', 'follow_params.yaml')

    collision_monitor = Node(
        package='nav2_collision_monitor',
        executable='collision_monitor',
        name='collision_monitor',
        parameters=[collision_params],
        output='screen')

    # smoother = Node(
    #     package='nav2_velocity_smoother',
    #     executable='velocity_smoother',
    #     name='velocity_smoother',
    #     parameters=[smoother_params],
    #     remappings=[('cmd_vel','/cmd_vel_collision'),
    #                 ('smoothed_cmd_vel','/cmd_vel')],
    #     output='screen')

    twist_stamper = Node(
            package='twist_stamper',
            executable='twist_stamper',
            remappings=[('/cmd_vel_in','/cmd_vel_collision'),
                        ('/cmd_vel_out','/diff_cont/cmd_vel')],
         )
         
    
    follower = Node(
        package='golftrolley',
        executable='mono_lidar_follower.py',
        name='mono_lidar_follower',
        output='screen',
        parameters=[follow_params],
    )



    # to_diff = Node(
    #     package='topic_tools', executable='relay', name='cmd_vel_to_diff',
    #     arguments=['/cmd_vel', '/diff_cont/cmd_vel'],  # simple relay
    #     output='screen'
    # )

    lifecycle_mgr = Node(
        package='nav2_lifecycle_manager',
        executable='lifecycle_manager',
        name='lifecycle_manager_safety',
        output='screen',
        parameters=[{
            'autostart': True,
            'bond_timeout': 0.0,
            'node_names': [ 'collision_monitor'] #'velocity_smoother',
        }]
    )

    yolo_v8 = IncludeLaunchDescription(
        PythonLaunchDescriptionSource(
            PathJoinSubstitution([
                FindPackageShare('yolo_bringup'),
                'launch',
                'yolov8.launch.py'
            ])
        ),
        launch_arguments={
            'input_image_topic': '/left/image_rect_rgb',   # matches yolo_bringup arg
            'model': 'yolov8n.pt',           # optional, defaults are fine
            'threshold': '0.35',
            # 'yolo_encoding': 'bgr8',       # only if your frames are bgr8
            'device': 'cpu',
        }.items()
    )

    yolo_delayed = TimerAction(period=4.0, actions=[yolo_v8])
    follower_delayed = TimerAction(period=15.0, actions=[follower])

    return LaunchDescription([
        # disparity_launch,
        collision_monitor,
        # cloud,
        # disparity,
        # point_cloud,
        lifecycle_mgr,
        twist_stamper,
        follower_delayed,
        yolo_delayed,
    ])
